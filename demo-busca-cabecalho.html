// Código Wix Velo (Adicione no masterPage.js para funcionar em todas as páginas)
import wixLocation from 'wix-location';

const ROUTES = [
  { url: '/newsletter', keywords: ['newsletter', 'boletim', 'artigo', 'edicao', 'publicacao', 'noticia'] },
  { url: '/webinares',  keywords: ['webinar', 'webinares', 'live', 'ao vivo', 'transmissao', 'debate', 'gravacao'] },
  { url: '/congressos', keywords: ['congresso', 'solaci', 'tct', 'sbhci', 'valves', 'evento', 'simposio'] },
  { url: '/cursos',     keywords: ['curso', 'cursos', 'modulo', 'capacitacao', 'intervencionista', 'aula'] }
];

$w.onReady(function () {
  const searchInput = $w('#headerSearchInput'); // ID do seu Input de busca nativo do Wix

  searchInput.onKeyPress((event) => {
    if (event.key === 'Enter') {
      const query = (searchInput.value || '').toLowerCase().trim();
      if (!query) return;

      // Encontra a rota com maior correspondência
      let matchedRoute = ROUTES.find(r => r.keywords.some(kw => query.includes(kw) || kw.includes(query)));
      
      if (matchedRoute) {
        wixLocation.to(matchedRoute.url);
      } else {
        // Redirecionamento padrão para congressos ou busca geral
        wixLocation.to('/congressos');
      }
    }
  });
});

// Código Wix Velo para Dropdown nativo ($w('#headerDropdown'))
import wixLocation from 'wix-location';

$w.onReady(function () {
  const dropdown = $w('#headerDropdown'); // ID do seu Dropdown nativo no Wix

  dropdown.options = [
    { label: "📰 Newsletters", value: "https://www.cathlabflix.org/newsletter" },
    { label: "🎥 Webinares",   value: "https://www.cathlabflix.org/webinares" },
    { label: "🏛️ Congressos",  value: "https://www.cathlabflix.org/congressos" },
    { label: "🎓 Cursos",      value: "https://www.cathlabflix.org/cursos" }
  ];

  dropdown.onChange((event) => {
    const selectedUrl = dropdown.value;
    if (selectedUrl) {
      wixLocation.to(selectedUrl);
    }
  });
});
